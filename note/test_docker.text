# airflow in docker

AIRFLOW_HOME=/opt/airflow
AIRFLOW__CORE__DAGS_FOLDER=/opt/airflow/dags
AIRFLOW__CORE__LOAD_EXAMPLES=False
AIRFLOW__CORE__EXECUTOR=
AIRFLOW__DATABASE__SQL_ALCHEMY_CONN=postgresql+psycopg2://airflow:airflowl123@postgres:5432/airflow_db
AIRFLOW__DATABASE__SQL_ALCHEMY_CONN_ASYNC=postgresql+asyncpg://airflow:airflowl123@postgres:5432/airflow_db



# docker image run with different command for different airflow service
#           Detached command production-style  -d = real service (production)
# API Server         -d (Detached mode)
docker run -d \
  --name airflow-api \
  -p 8080:8080 \
  -e AIRFLOW__DATABASE__SQL_ALCHEMY_CONN=postgresql+psycopg2://airflow:airflowl123@host.docker.internal:5432/airflow_db \
  deathsamsul/airflow-core:3.2.0 \
  airflow api-server

# Scheduler
docker run -d \
  --name airflow-scheduler \
  deathsamsul/airflow-core:3.2.0 \
  airflow scheduler

# DAG Processor
docker run -d \
  --name airflow-dag-processor \
  deathsamsul/airflow-core:3.2.0 \
  airflow dag-processor



apiVersion: apps/v1
kind: Deployment
metadata:
  name: airflow-worker
spec:
  replicas: 1
  selector:
    matchLabels:
      app: airflow-worker
  template:
    metadata:
      labels:
        app: airflow-worker
    spec:
      containers:
        - name: worker
          image: airflow-worker:3.1
          imagePullPolicy: Never
          command: ["airflow", "celery", "worker"]
          env:
            - name: AIRFLOW__CORE__EXECUTOR
              value: CeleryExecutor
            - name: AIRFLOW__CELERY__BROKER_URL
              value: redis://redis:6379/0
            - name: AIRFLOW__DATABASE__SQL_ALCHEMY_CONN
              value: postgresql+psycopg2://airflow:airflow@postgres:5432/airflow
            - name: PYTHONPATH
              value: /opt/airflow:/opt/airflow/app
          volumeMounts:
            - name: app-code
              mountPath: /opt/airflow/app
            - name: dags-code
              mountPath: /opt/airflow/dags
      volumes:
        - name: app-code
          hostPath:
            path: /home/sam/projects/fraud_detection_mlops/app
            type: Directory
        - name: dags-code
          hostPath:
            path: /home/sam/projects/fraud_detection_mlops/app/airflow_dags
            type: Directory
